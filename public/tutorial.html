2. การจัดการฐานข้อมูล (Database Management)

ทำความเข้าใจไฟล์ server/database.py และตารางหลักในระบบ

ระบบใช้ pymysql ในการเชื่อมต่อฐานข้อมูล TiDB Cloud (ซึ่งเข้ากันได้สมบูรณ์กับ MySQL Driver) และมี Custom Wrapper ในการแปลง SQL Placeholders

SSL Connection Pool

เชื่อมต่อผ่านพอร์ต 4000 พร้อมเปิดใช้งาน SSL Certificate certifi.where() เพื่อความปลอดภัยสูงสุดขณะส่งข้อมูลข้าม Cloud

Custom Cursor Wrapper

คลาส MySQLCursorWrapper จะคอยแทนที่เครื่องหมาย ? ให้กลายเป็น %s อัตโนมัติ ทำให้เขียน Query ง่ายและป้องกัน SQL Injection

server/database.py Python 3.11 
class MySQLCursorWrapper:
    def __init__(self, cursor):
        self.cursor = cursor

    def execute(self, query, args=None):
        # แปลง SQLite placeholder '?' ให้เป็น MySQL '%s' อัตโนมัติ
        if args is not None:
            query = query.replace('?', '%s')
            return self.cursor.execute(query, args)
        return self.cursor.execute(query)

def get_db_connection():
    # ดึงค่าการเชื่อมต่อจากไฟล์ .env
    host = os.getenv("TIDB_HOST")
    user = os.getenv("TIDB_USER")
    password = os.getenv("TIDB_PASSWORD")
    database = os.getenv("TIDB_DATABASE")
    
    conn = pymysql.connect(
        host=host, port=4000, user=user, password=password, database=database,
        cursorclass=pymysql.cursors.DictCursor, ssl={'ca': certifi.where()}
    )
    return MySQLConnectionWrapper(conn)

3. ระบบยืนยันตัวตนและการรักษาความปลอดภัย (Auth & JWT Flow)

เจาะลึกไฟล์ server/auth.py และการทำงานของ JSON Web Tokens

  • 1

    การเข้ารหัสรหัสผ่าน (Password Hashing with Salt)

    ใช้สุ่ม secrets.token_hex(16) เป็น Salt แล้วนำรหัสผ่านเข้ารหัสด้วย PBKDF2 HMAC SHA-256 ซ้ำ 100,000 รอบก่อนเซฟลงตาราง users

  • 2

    การเข้าสู่ระบบและการรับ JWT Token (Login Step)

    เมื่อเข้าสู่ระบบสำเร็จ ฟังก์ชัน create_access_token() จะสร้าง JWT Token ที่ลงลายมือชื่อดิจิทัลด้วย SECRET_KEY และมีอายุการใช้งาน 7 วัน

  • 3

    การยืนยันตัวตนในทุก Request (Token Verification)

    Frontend จะแนบ Authorization: Bearer <token> มาใน HTTP Header ฟังก์ชัน get_current_user() จะแกะ Token เพื่อยืนยันว่าผู้ใช้คือใครก่อนประมวลผล

server/auth.py JWT & Password Hashing 
# 1. ฟังก์ชันสุ่ม Salt และ Hash รหัสผ่าน
def get_password_hash(password: str) -> str:
    salt = secrets.token_hex(16)
    hash_obj = hashlib.pbkdf2_hmac(
        'sha256', password.encode('utf-8'), salt.encode('utf-8'), 100000
    )
    return f"{salt}:{hash_obj.hex()}"

# 2. ฟังก์ชันสร้าง JWT Access Token
def create_access_token(data: dict, expires_delta: Optional[timedelta] = None) -> str:
    to_encode = data.copy()
    expire = datetime.now(timezone.utc) + (expires_delta or timedelta(days=7))
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, SECRET_KEY, algorithm="HS256")

4. การส่ง HTTP API และการจัดการ State ฝั่ง Frontend

อธิบายการทำงานของ React AuthContext.tsx และการยิง API ด้วย fetch()

ฝั่ง Frontend ใช้ React Context (AuthContext) ในการเก็บข้อมูลผู้ใช้งานและ Token ไว้ในหน่วยความจำและ localStorage

client/pages/Login.tsx TypeScript / React 
const handleLogin = async () => {
    // 1. ส่ง request ไปที่ Backend API
    const res = await fetch("/api/auth/login", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email, password })
    });
    
    const data = await res.json();
    if (res.ok) {
        // 2. บันทึก Token ลง localStorage และเซ็ต State
        localStorage.setItem("token", data.access_token);
        login(data.access_token, data.user);
        toast.success("เข้าสู่ระบบสำเร็จ!");
    }
};
server/routes/auth_routes.py FastAPI Endpoint 
@router.post("/login")
async def login(user_credentials: UserLogin):
    conn = get_db_connection()
    cursor = conn.cursor()
    
    # 1. ค้นหา User จากฐานข้อมูล
    cursor.execute("SELECT * FROM users WHERE email = ?", (user_credentials.email,))
    user = cursor.fetchone()
    
    # 2. ตรวจสอบรหัสผ่าน
    if not user or not verify_password(user_credentials.password, user['password']):
        raise HTTPException(status_code=401, detail="อีเมลหรือรหัสผ่านไม่ถูกต้อง")
        
    # 3. ออก JWT Token
    access_token = create_access_token(data={"sub": user['email']})
    return {"access_token": access_token, "token_type": "bearer", "user": user}

  1. ระบบดึงข้อมูลคำถาม เฉลย และ Rubrics JSON จากตาราง exam_questions
  2. สร้าง Prompt กำหนดให้ Gemini ตอบกลับมาในรูปแบบ Structured JSON Schema เท่านั้น
